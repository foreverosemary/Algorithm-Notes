### 点/向量模板

- 用 $*$ 表示向量点积，^ 表示向量叉积。

```cpp
// 点结构体
struct Point {
    double x, y;
    Point(){}
    Point(double x, double y) : x(x), y(y) {}
    bool operator ==(Point p) const {return sgn(x - p.x) == 0 && sgn(y - p.y) == 0;}
    void input(){cin >> x >> y;}
    bool operator <(Point p) const {return sgn(x - p.x) < 0 || (sgn(x - p.x) == 0 && sgn(y - p.y) < 0);}
    // 向量加法
    Point operator +(Point p) const {return Point(x + p.x, y + p.y);}
    // 向量减法
    Point operator -(Point p) const {return Point(x - p.x, y - p.y);}
    // 点积（内积）：向量A在向量B上的投影长度乘向量B的模长
    double operator *(Point p) const {return x * p.x + y * p.y;}
    // 数乘
    Point operator *(double k) const {return Point(x * k, y * k);}
    Point operator /(double k) const {return Point(x / k, y / k);}
    // 叉积（外积）：向量A与B张成的平行四边形的有向面积
    double operator ^(Point p) const {return x * p.y - y * p.x;}
    // 距离（模长）：向量A到向量B的有向距离
    double len(Point p = Point(0, 0)) const {return hypot(x - p.x, y - p.y);}
    // 角度（弧度制）：两向量无向夹角[0, π]
    double angle(Point p = Point(1, 0)) const {return acos(*this * p / len() / p.len());}
    // 角度（弧度制）：两向量有向夹角[-π, π]
    double dangle(Point p = Point(1, 0)) const {
        double delta = atan2(p.x, p.y) - atan2(x, y);
        while(delta > PI) delta -= 2 * PI;
        while(delta < -PI) delta += 2 * PI;
        return delta;
    }
    // 旋转（弧度制）：逆时针旋转theta弧度
    Point rotate(double theta) {return Point(x * cos(theta) - y * sin(theta), y * cos(theta) + x * sin(theta));}
};
```

### 线/线段模板

```cpp
// 线段与直线结构体
struct Line {
    Point s, e;
    double k; // 极角：
    Line() {}
    Line(Point s, Point e) : s(s), e(e) {k = angle();}
    Line(double x1, double y1, double x2, double y2) {s = Point(x1, y1), e = Point(x2, y2); k = angle();}
    // 点斜式
    Line(Point p, double k_) : s(p), k(atan2(k_, 1)) {
        if(sgn(k - PI / 2) == 0) e = s + Point(0, 1);
        else e = s + Point(1, tan(k));
    }
    // 一般式：ax+by+c=0
    Line(double a, double b, double c) {
        if(sgn(a) == 0) {s = Point(0, -c / b), e = Point{1, -c / b};}
        else if(sgn(b) == 0) {s = Point(-c / a, 0), e = Point{-c / a, 1};}
        else {s = Point(0, -c / b), e = Point(1, (-c - a) / b);}
        k = angle();
    }
    void input(){s.input(); e.input(); k = angle();}
    void adjust(){if(e.y < s.y) swap(s, e);}
    bool operator <(Line l) {return sgn(k - l.k) < 0 || (sgn(k - l.k) == 0 && l.relation(s) < 0);}
    // 线段长度
    double len() {return s.len(e);}
    // 直线倾斜角(-π, π]
    double angle() {return atan2(e.y - s.y, e.x - s.x);}
    // 点和直线的关系：负表左，正表右
    int relation(Point p) {return sgn((p - s) ^ (e - s));}
    // 判断点是否在线段上
    bool onseg(Point p) {return relation(p) == 0 && sgn((p - s) * (e - s)) <= 0;}
    // 判断两直线是否平行
    bool parallel(Line l) {return sgn((e - s) ^ (l.e - l.s)) == 0;}
    // 判断两线段关系
    bool segcroseg(Line l) {
        adjust(), l.adjust();
        // 快速排斥
        if(e < l.s || l.e < s || sgn(min(l.s.y, l.e.y) - max(s.y, e.y)) > 0 
        || sgn(min(s.y, e.y) - max(l.s.y, l.e.y)) > 0) return false;
        // 跨立实验
        return relation(l.e) * relation(l.s) <= 0 && l.relation(s) * l.relation(e) <= 0;
    }
    // 直线与线段的关系
    bool linecroseg(Line l) {
        adjust(), l.adjust();
        int d1 = sgn((l.s - s) ^ (l.e - s));
        int d2 = sgn((l.s - e) ^ (l.e - e));
        return d1 * d2 <= 0;
    }
    // 求两直线交点
    Point linecroline(Line l) {
        Point v1 = e - s, v2 = l.e - l.s;
        double t = ((l.s - s) ^ v2) / (v1 ^ v2);
        return s + v1 * t;
    }
    // 求中垂线
    Line midline() {return Line((s + e) * 0.5, (s - e).rotate(PI / 2));}
};
```

```cpp
struct trangle {
    Point a, b, c;
    double la, lb, lc;
    trangle(){}
    trangle(Point a, Point b, Point c) : a(a), b(b), c(c) {la = a.len(b), lb = b.len(c), lc = c.len(a);}
    trangle(double la, double lb, double lc) : la(la), lb(lb), lc(lc) {}
    trangle(double x1, double y1, double x2, double y2, double x3, double y3) {
        a = Point(x1, y1), b = Point(x2, y2), c = Point(x3, y3);
        la = a.len(b), lb = b.len(c), lc = c.len(a);
    }
    // 三角形面积
    double area() {return fabs((b - a) ^ (c - a)) * 0.5;}
    // 三角形周长
    double len() {return la + lb + lc;}
    // 外接圆半径
    double outr() {return la * lb * lc / (4 * area());}
    // 内切圆半径
    double inr() {return area() * 3 / (la + lb + lc);}
};
```

```cpp
// 圆结构体
struct Circle {
    Point o;
    double r;
    Circle() {}
    Circle(Point o, double r) : o(o), r(r) {}
    double area() {return PI * r * r;}
    // 点与圆的关系
    int relation(Point p) {
        double len = p.len(o);
        if(sgn(len - r) < 0) return -1;
        else if(sgn(len - r) == 0) return 0;
        else return 1;
    }
    // 圆与圆的关系
    int relation(Circle c) {
        double len = o.len(c.o);
        if(sgn(len - r - c.r) < 0) return -1;
        else if(sgn(len - r - c.r) == 0) return 0;
        else return 1;
    }
};
```

# 二、快速排斥与跨立实验

- 快速排斥实验：规定 **一段线段的区域** 为以这条线段为对角线，各边与某一坐标轴平行的矩形所占的区域。易得，**如果两个线段没有公共区域，那么这两个线段一定不相交**。判断一个点 $p$ 是否在一段线段 $(s,e)$ 的区域，若 $(s-p)*(s-e)\le0$，代表该点在该区域内，否则在区域外。
- 跨立实验：如果两个线段 $a,b$ 相交，那么 $b$ 线段的两个端点一定分布在 $a$ 线段的两侧；对于 $a$ 也同理。分别计算一个线段的两点与另外一个线段的向量积，判断两点与该线段的位置关系。如果 $a$ 的两端点在 $b$ 的两侧，且 $b$ 的两端点在 $a$ 的两侧，说明 $a,b$ 两个线段通过了跨立实现。

由于两线段不相交但共线的特殊情况会通过跨立实验，**当两线段通过了快速排斥实验再通过跨立实验时，说明两线段相交**。

# 三、多边形

### 点与多边形的关系（PIP 问题）

1. 射线法
从测试点出发，向任意方向（通常取无理数避免与多边形某条边重合）发射一条射线吗，统计该射线与多边形边界的交点数量：
- 若交点数量为 **奇数**，说明测试点在多边形内部；
- 若交点数量为 **偶数**，说明测试点在多边形外部；
简记为 **奇内偶外**。

2. 绕组数法
将该点与多边形的所有顶点相连，计算相邻两边有向角之和，如果和为 $0$ 代表点在多边形外，否则在多边形内部。

---
### 多边形的面积

将多边形上的点逆时针标记为 $p_1,p_2,\dots,p_n$，任选辅助点 $O$，取向量 $v_i=p_i-O$，那么该多边形被分解为了一个个的小三角形 $p_iOp_{(i\ mod\ n)+1}$，那么该多边形面积 $S$ 可表示为所有小三角形面积之和：
$$
\begin{aligned}
S=\frac{1}{2}|\sum_{i=1}^{n}v_{i}\times v_{(i\ mod\ n)+1}|
\end{aligned}
$$
---
## 凸包

- 凸多边形：指所有内角大小都在 $[0,\pi]$ 范围内的简单多边形。
- 凸包：在平面上能够包含所有给定点的最小凸多边形叫做凸包。
凸包用最小的周长围住了给定的所有点，如果一个凹多边形围住了所有的点，它的周长一定不是最小的。

### Andrew 算法

算法思路：
1. 将所有点以横坐标为第一关键字，纵左边为第二关键字排序。
2. 排序后最小的元素和最大的元素一定在凸包上。对于凸多边形，从一个点出发逆时针走，轨迹总是左拐的，如果出现右拐，那么这一段不在凸包上。因此用一个单调栈维护上下凸壳。从左向右看，由于上下凸壳的旋转方向不同，应改先升序枚举求出下凸壳，再降序求出上凸壳。
3. 求凸壳时，一旦发现当前枚举到的点 $P$ 和栈顶的两个点 $S_1,S_2$（$S_1$ 为栈顶的点）行进的方向向右旋转，即 $\overrightarrow{S_1P}\times\overrightarrow{S_2S_1}\lt0$，则弹出栈顶，直到 $\overrightarrow{S_1P}\times\overrightarrow{S_2S_1}\ge0$ 或栈中仅一个元素为止。（如果需要保留位于凸包边上的点，则 $\overrightarrow{S_1P}\times\overrightarrow{S_2S_1}\lt0$ 中的 $\lt$ 改为 $\le$，同时后面的条件也相应改变为 $\gt$）。

```cpp
const int N;
int stk[N], tp, n;
Point[N];

void andrew() {
	sort(p + 1, p + 1 + n);
	stk[++tp] = 1;
	for(int i = 2; i <= n; i++) {
		while(tp >= 2 && ((p[stk[tp]] - p[stk[tp - 1]]) ^ (p[i] - p[stk[tp]])) <= 0) 
			tp--;
		stk[++tp] = i;
	}
	int tmp = tp;
	for(int i = n - 1; i > 0; i--) {
		if(p[stk[tp]] == p[i]) continue;
		while(tp > tmp && ((p[stk[tp]] - p[stk[tp - 1]]) ^ (p[i] - p[stk[tp]])) < 0)
			tp--;
		stk[++tp] = i;
	}
	double ans = 0;
	for(int i = 1; i < tp; i++) {
		ans += p[stk[i]].len(p[stk[i + 1]]);
	}
	cout << fixed << setprecision(2) << ans << endl;
}
```

# 四、二维距离

## 曼哈顿距离转切比雪夫距离

在曼哈顿坐标系下对于两点 $A(x_1,y_1),B(x_2, y_2)$，有：
$$
d(A, B)\ =\ |x_1-x_2|+|y_1-y_2|=max(|(x_1+y_1)-(x_2+y_2)|,|(x_1-y_1)-(x_2-y_2)|)
$$
这就是 $(x_1+y_1, x_1-y_1),(x_2+y_2,x_2-y_2)$ 两点之间的切比雪夫距离。

所以将每一个点 $(x, y)$ 转化为 $(x+y, x-y)$ ，新坐标系下的切比雪夫距离为原坐标系下的曼哈顿距离。

---
## 切比雪夫距离转曼哈顿距离

同理，$A,B$ 两点的切比雪夫距离为：
$$
d(A,B)\ =\ max(|x_1-x_2|,|y_1-y_2|)\ =\ |\frac{x_1+y_1}{2}-\frac{x_2+y_2}{2}|+|\frac{x_1-y_1}{2}-\frac{x_2-y_2}{2}|
$$
这就是 $(\frac{x_1+y_1}{2},\frac{x_1-y_1}{2}),(\frac{x_2+y_2}{2},\frac{x_2-y_2}{2})$ 两点之间的曼哈顿距离。

***注意：*** **切比雪夫距离** 转成 **曼哈顿距离** 时要除以 $2$ ，为了避免出现小数，我们可以横坐标和纵坐标同时乘上 $2$ ，最后答案除以 $2$ 。

---
## 最小曼哈顿距离之和

考虑一组点集 $\{A_1,A_2,\dots,A_n\}$ ，坐标为 $(x_1,y_1),(x_2,y_2),\dots,(x_n,y_n)$ ，找出一个点，使得该点到所有点的曼哈顿距离之和最小。

枚举所选点 $i$，设前 $k$ 个点的横坐标都 $\leq x_i$ ，计算其它点到它的曼哈顿距离和为：
$$
d\ =\ kx_i-\sum_{i=1}^{k}x_i+\sum_{i=k+1}^{n}x_i-(n-k)x_i\ =\ (2k-n)x_i+prefix[n]-2\times prefix[k]
$$
那么只需要将横坐标升序排序后二分查找即可，对于纵坐标同理。

# 五、半平面交

- 半平面：一条直线和直线的一侧，本质是一个点集。当包含直线时，称为闭半平面；不包含直线时，称为开半平面。解析式一般用 $y\ge kx+b$（左半平面）或 $y\le kx+b$（右半平面）表示。
- 半平面交：是指多个半平面的交集，在平面直角坐标系围成一个区域，常取左半平面交。可应用于求凸多边形的面积交，以及线性规划的可行域

## S&I 算法（排序增量法）

算法思路：
- 极角排序：对于多边形上的边对应的向量构成的边集，以 $\theta=arctan{\frac{y}{x}}$ 为关键字进行排序（即极角排序）。如果遇到共线且方向相同的向量，取靠近可行域的一个（比如两个向量的极角相同，而我们要的是向量的左侧半平面，那么我们只需要保留左侧的向量）。
- 维护单调队列：由于半平面是一个凸多边形，所以需要维护一个凸壳。由于新加入的边只会影响最开始加入或最后加入的边，只需要删除队首或者队尾的元素即可。遍历排序后的向量数组时，维护一个交点数组，当单调队列中的向量超过 $2$ 个时，会产生交点，对于当前遍历到的变量，如果上一个交点在这个向量表示的半平面交的异侧，那么上一个向量需要去除。

```cpp
const int N;
Line lines[N]; 
int n;

double S_L() {
	sort(lines + 1, lines + 1 + n);
    vector<Line> q(n + 1); // 双端队列
    int h = 1, t = 1;
    q[1] = lines[1];
    for(int i = 2; i <= n; i++) {
        if(sgn(lines[i].k - lines[i - 1].k) == 0) continue;
        while(h < t && lines[i].relation(q[t].linecroline(q[t - 1])) >= 0) t--;
        while(h < t && lines[i].relation(q[h].linecroline(q[h + 1])) >= 0) h++;
        q[++t] = lines[i];
    }
    while(h < t - 1 && q[h].relation(q[t].linecroline(q[t - 1])) >= 0) t--;
    q[++t] = q[h];
    double ans = 0;
    vector<Point> p;
    for(int i = h; i < t; i++) p.push_back(q[i].linecroline(q[i + 1]));
    for(int i = 1; i + 1 < p.size(); i++) ans += fabs((p[0] - p[i]) ^ (p[i] - p[i + 1]));
    return ans / 2;
}
```

## 平面中的可见性

- 多边形的内核（也常被称为核）：考虑一个点集 $P$，对于点集中的每个点 $p_i\in P$，满足所有位于多边形 $T$ 上的点 $p_j\in T$，使得过这两个点的射线 $p_ip_j$ 与多边形的交点仅有 $p_j$，这个点集 $P$ 被称为多边形的核，完全位于多边形的内部。

从另一个角度来看，位于核的位置可以看到多边形的任何一个地方。将 $T$ 的边转化为内部方向的半平面，每条边对应一个可见性约束，求这些边的半平面交就是求 $T$ 的核。如果给定平面点集 $p_1,p_2,\dots,p_n$，判断是否存在一个点 $q$，使得从 $q$ 看这些点时，顺序能够为 $p_1,p_2,\dots,p_n$（即 $p_i$ 在 $p_j$ 观察顺序上的左侧，$i\lt j$），对于连续点 $p_i$ 和 $p_{i+1}$ 构成的直线定义为一个半平面，判断 $q$ 点是否存在即为这些直线的半平面交是否为空。

# 六、最小圆覆盖

在一个平面上有 $n$ 个点，求一个半径最小的圆，能覆盖到所有的点。

## 随机增量法

- 增量法：思想与第一数学归纳法类似，将问题拆分为规模刚好小一层的子问题，解决子问题后加入当前的对象：$T(n)=T(n-1)+g(n)$。结合随机化可避免最坏情况出现，时间复杂度通常为 $O(n)$，最坏情况下为 $O(n^2)$。

```cpp
const int N;
int n;
Point p[N];
Circle c;

double sqr(double x) {return x * x;}

Point geto(Point a, Point b, Point c) {
    double a1 = 2 * (b.x - a.x), b1 = 2 * (b.y - a.y);
    double a2 = 2 * (c.x - a.x), b2 = 2 * (c.y - a.y);
    double c1 = sqr(b.x) - sqr(a.x) + sqr(b.y) - sqr(a.y);
    double c2 = sqr(c.x) - sqr(a.x) + sqr(c.y) - sqr(a.y);
    if(sgn(a1) == 0) return Point((c2 - c1 * b2 / b1) / a2, c1 / b1);
    if(sgn(b1) == 0) return Point(c1 / a1, (c2 - c1 * a2 / a1) / b2);
    double x = (c1 * b2 - c2 * b1) / (a1 * b2 - a2 * b1);
    double y = (c1 * a2 - c2 * a1) / (a2 * b1 - a1 * b2);
    return Point(x, y);
}

int main() {
    // 随机化
    for(int i = 1; i <= n; i++) swap(p[rand() % n + 1], p[rand() % n + 1]);
    c = {p[1], 0};
    for(int i = 2; i <= n; i++) {
        if(c.relation(p[i]) <= 0) continue;
        c = {p[i], 0};
        for(int j = 1; j < i; j++) {
            if(c.relation(p[j]) <= 0) continue;
            c = {(p[i] + p[j]) / 2, p[i].len(p[j]) / 2};
            for(int k = 1; k < j; k++) {
                if(c.relation(p[k]) <= 0) continue;
                c.o = geto(p[i], p[j], p[k]);
                c.r = p[i].len(c.o);
            }
        }
    }
    return 0;
}
```

# 七、平面最近点对

在一个平面上有 $n$ 个点，求一组欧几里得距离最近的点对。

## 分治算法

- 先将所有的点按照 $x$ 为第一关键字、$y$ 为第二关键字进行排序，并以 $p_{mid}\big( mid = \lfloor \frac{n}{2}\rfloor \big)$ 为分界点，拆分点集为 $A_1,A_2$，并递归求出两个点集各自内部的最近点对，距离为 $h_1,h_2$，取最小值为 $h$。
- 递归完成后合并两个点集，即找到一个点属于 $A_1$，另一个点属于 $A_2$，且两者距离小于 $h$。因此先将所有横坐标与 $x_{mid}$ 的差小于 $h$ 的点放入集合 $B = \{ p _ { i } \big| |x _ { i } - x _ { m } | < h \}$，对于集合 $B$ 中的点 $p_i$，考虑纵坐标小于 $y_i$ 并且纵向距离不超过 $h$ 的点，构成集合 $C(P_i)=\{p_j\big|p_{j}\in B,y_{i}-h\lt y_{j}\lt y_i\}$，对于每对 $(p_i,p_j)$ 计算距离并更新答案。

```cpp
const int N;
int n;
Point p[N], rem[N];

double solve(int l, int r) {
    if(l >= r) return 2e9;
    if(l + 1 == r) return p[l].len(p[r]);
    int mid = (l + r) >> 1;
    double d = min(solve(l, mid), solve(mid + 1, r));
    int tot = 0;
    for(int i = l; i <= r; i++) 
        if(sgn(fabs(p[i].x - p[mid].x) - d) < 0) rem[++tot] = p[i];
    sort(rem + 1, rem + 1 + tot, [&](const auto &a, const auto &b) {
        return a.y < b.y;
    });
    for(int i = 1; i <= tot; i++) {
        for(int j = i - 1; j >= 1 && sgn(rem[i].y - rem[j].y - d) < 0; j--) {
            d = min(d, rem[i].len(rem[j]));
        }
    }
    return d;
}

int main() {
    sort(p + 1, p + 1 + n);
    double ans = solve(1, n);
    return 0;
}
```

## 非分治算法

- 先将所有点按照 $x$ 为第一关键字进行排序，并维护一个以 $y$ 为第二关键字进行排序的  `multiset`，用于存储可能与当前点产生最近距离的点，取 $d$ 表示最近点对距离。
- 对于当前遍历到的点，先移除多重集中满足 $x_i-x_{j}\ge d$ 的点，再调用二分查找多重集中满足 $y_i-y_{j}\lt d$ 的点，计算每对 $(p_i,p_j)$ 的距离并更新答案。

```cpp
const int N;
Point p[N];
int n;

struct cmpy {
    bool operator ()(const Point &a, const Point &b) const {return a.y < b.y;}
};

struct cmpx {
    bool operator ()(const Point &a, const Point &b) const {return a.x < b.x;}
};

double solve() {
    double d = 2e9;
    multiset<Point, cmpy> s;
    sort(p, p + n, cmpx());
    for(int i = 0, l = 0; i < n; i++) {
        while(l < i && sgn(p[i].x - p[l].x - d) >= 0) s.erase(s.find(p[l++]));
        for(auto it = s.lower_bound(Point(p[i].x, p[i].y - d));
            it != s.end() && sgn(it->y - p[i].y - d) < 0; it++)
        d = min(d, p[i].len(*it));
        s.insert(p[i]);
    }
    return d;
}
```

# 八、旋转卡壳

- 切线：给定一个凸多边形，切线是一条与凸多边形相交的直线，且凸多边形上的所有点都在切线的同一侧。
- 对踵点对：如果凸多边形的两个点 $p$ 和 $q$ 在两条平行切线上，那么它们就构成一对 **对踵点对**。根据切线与凸多边形的相交方式，分为”点-点“对踵点对、”边-点“对踵点对（即过某点的切线与凸多边形平行）、”边-边“对踵点对，凸 $n$ 边形的对踵点对最多有 $\frac{3n}{2}$ 对。

在调用 Andrew 算法求解完凸包后，以求解凸包直径为例展示求解某条边对应的对踵点：

```cpp
const int N;
Point p[N];
int tp, n, stk[N];

double corss(Point a, Point b, Point c) {
	return fabs((b - a) ^ (c - b));
}

void andrew() {
	sort(p + 1, p + 1 + n);
	stk[++tp] = 1;
	for(int i = 2; i <= n; i++) {
		while(tp >= 2 && ((p[stk[tp]] - p[stk[tp - 1]]) ^ (p[i] - p[stk[tp]])) <= 0) 
			tp--;
		stk[++tp] = i;
	}
	int tmp = tp;
	for(int i = n - 1; i > 0; i--) {
		if(p[stk[tp]] == p[i]) continue;
		while(tp > tmp && ((p[stk[tp]] - p[stk[tp - 1]]) ^ (p[i] - p[stk[tp]])) < 0)
			tp--;
		stk[++tp] = i;
	}
}

void solve() {
	andrew(); // 求解凸包
	tq--; // 
	double ans = 0;
	for(int i = 1, j = 2; i <= tp; i++) {
		while(cross(p[stk[i]], p[stk[i + 1]], p[stk[j]]) <
		cross(p[stk[i]], p[stk[i + 1]], p[stk[j + 1]])) j = j % tp + 1;
		ans = max(ans, max(p[stk[i]].len(p[stk[j]], p[stk[i + 1]].len(p[stk[j]])));
	}
	cout << ans << endl;
}
```

# 九、闵可夫斯基和

点集 $P$ 和点集 $Q$ 的闵可夫斯基和 $P+Q$ 定义为 $$P+Q=\{a+b\mid a\in P,b\in Q\}$$即把点集 $Q$ 中的每个点看做一个相连，将点集 $P$ 中的每个点向这些向量平移，最终得到的结果的集合就是点集 $P+Q$。

**性质**：
- 若点集 $P,Q$ 为凸集，则其闵可夫斯基和 $P+Q$ 也为凸集。
- 若点集 $P,Q$ 为凸集，则其闵可夫斯基和 $P+Q$ 的边集是由凸集 $P,Q$ 的边按极角排序后连接的结果。
